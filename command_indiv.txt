(
  set -e
  HRM_RUN_ID="20261014"

  for HRM_INPUT in cable_loadcell; do
    bash scripts/hrm_python.sh train.py \
      --config "configs/train_${HRM_INPUT}.yaml" \
      --output "results/train/${HRM_RUN_ID}_${HRM_INPUT}"

    bash scripts/predict_and_organize.sh \
      "results/train/${HRM_RUN_ID}_${HRM_INPUT}" \
      "results/predict/${HRM_RUN_ID}_${HRM_INPUT}" \
      --id-tolerance 2
  done
)
